Configuration du Service Worker

Centralisez vos accès YouTube et Twitch/Kick via un seul Cloudflare Worker sécurisé.

Accéder à la page d'administration →

Guide Cloudflare Unifié

1. Configuration Cloudflare

  1. Créez un Worker nommé omnimark-worker.
  2. KV : Créez un namespace OMNIMARK_KV et liez-le au worker (Variable = OMNIMARK_KV).
  3. Déclencheurs : Ajoutez un Cron Trigger toutes les 5 minutes (*/5 * * * *).
  4. Variables d'environnement : Ajoutez uniquement API_KEY (votre clé secrète ci-dessus). Les autres clés se configurent sur la page /admin du worker (YouTube est optionnel).

2. Code du Worker Unifié

const getConfig = async (env, key) => {
  const val = await env.OMNIMARK_KV?.get(key);
  return val || env[key];
};

const refreshCache = async (env) => {
  const tId = await getConfig(env, 'TWITCH_CLIENT_ID');
  const tSec = await getConfig(env, 'TWITCH_CLIENT_SECRET');
  const twitchChannelsRaw = await env.OMNIMARK_KV?.get('CHANNELS_TWITCH');
  const twitchChannels = twitchChannelsRaw ? JSON.parse(twitchChannelsRaw) : [];
  
  const ua = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/110.0.0.0 Safari/537.36";

  let twitchStreams = [];
  if (tId && tSec && twitchChannels.length > 0) {
    try {
      const authRes = await fetch(`https://id.twitch.tv/oauth2/token?client_id=${tId}&client_secret=${tSec}&grant_type=client_credentials`, { method: 'POST' });
      const { access_token } = await authRes.json();
      const names = twitchChannels
        .map(c => typeof c === 'string' ? c : (c.platform === 'twitch' || c.platform === 'both' ? c.name : null))
        .filter(n => n);

      if (names.length > 0) {
        const params = names.map(n => `user_login=${encodeURIComponent(n)}`).join('&');
        const res = await fetch(`https://api.twitch.tv/helix/streams?${params}`, { 
          headers: { 'Client-ID': tId, 'Authorization': `Bearer ${access_token}` } 
        });
        const { data } = await res.json();
        twitchStreams = (data || []).map(s => ({ 
          ...s, 
          platform: 'twitch', 
          thumbnail_url: s.thumbnail_url.replace('{width}', '440').replace('{height}', '248') 
        }));
      }
    } catch (e) {}
  }

  let kickStreams = [];
  const kickNames = twitchChannels
    .filter(c => typeof c === 'object' && (c.platform === 'kick' || c.platform === 'both'))
    .map(c => c.name);
    
  if (kickNames.length > 0) {
    const results = await Promise.all(kickNames.map(async (name) => {
      try {
        const res = await fetch(`https://kick.com/api/v1/channels/${encodeURIComponent(name.toLowerCase())}`, { 
          headers: { "User-Agent": ua } 
        });
        if (!res.ok) return null;
        const data = await res.json();
        if (data.livestream) {
          return { 
            user_login: name, 
            user_name: data.user?.username || name, 
            title: data.livestream.session_title, 
            game_name: (data.livestream.categories?.[0]?.name) || (data.recent_categories?.[0]?.name) || 'Kick', 
            viewer_count: data.livestream.viewer_count, 
            thumbnail_url: data.livestream.thumbnail?.url || data.user?.profile_pic || '', 
            platform: 'kick' 
          };
        }
      } catch (e) {}
      return null;
    }));
    kickStreams = results.filter(r => r !== null);
  }
  await env.OMNIMARK_KV?.put('CACHE_STREAMS', JSON.stringify([...twitchStreams, ...kickStreams]), { expirationTtl: 300 });

  const youtubeChannelsRaw = await env.OMNIMARK_KV?.get('CHANNELS_YOUTUBE');
  const youtubeChannels = youtubeChannelsRaw ? JSON.parse(youtubeChannelsRaw) : [];
  if (youtubeChannels.length > 0) {
    const results = await Promise.all(youtubeChannels.map(async (c) => {
      try {
        const res = await fetch(`https://www.youtube.com/feeds/videos.xml?channel_id=${c.id}`, { headers: { "User-Agent": ua } });
        const text = await res.text();
        const entries = text.split('<entry>').slice(1, 3);
        for (const entry of entries) {
          const vId = entry.match(/<yt:videoId>([^<]+)<\/yt:videoId>/)?.[1];
          const vTitle = entry.match(/<title>([^<]+)<\/title>/)?.[1];
          const vLink = entry.match(/<link rel="alternate" href="([^"]+)"/)?.[1];
          if (!vId || vTitle?.toLowerCase().includes('shorts') || vLink?.includes('/shorts/')) continue;
          
          const check = await fetch(`https://www.youtube.com/shorts/${vId}`, { method: 'HEAD', redirect: 'manual', headers: { "User-Agent": ua } });
          if (check.status === 200) continue;

          return { 
            id: vId, 
            title: vTitle, 
            author: entry.match(/<name>([^<]+)<\/name>/)?.[1], 
            channelId: c.id, 
            published: entry.match(/<published>([^<]+)<\/published>/)?.[1] || new Date().toISOString(), 
            thumbnail: entry.match(/<media:thumbnail url="([^"]+)"/)?.[1],
            link: `https://www.youtube.com/watch?v=${vId}` 
          };
        }
      } catch (e) {}
      return null;
    }));
    await env.OMNIMARK_KV?.put('CACHE_YOUTUBE', JSON.stringify(results.filter(r => r)), { expirationTtl: 900 });
  }
};

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const path = url.pathname;
    const apiKey = request.headers.get('X-API-Key') || url.searchParams.get('key');

    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type, X-API-Key",
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

    // --- Admin & Config ---
    if (path === '/admin') {
      const isAuthorized = !env.API_KEY || apiKey === env.API_KEY;

      if (request.method === 'POST') {
        if (!isAuthorized) return new Response("Unauthorized", { status: 401 });
        const data = await request.json();
        for (const [k, v] of Object.entries(data)) {
          if (['TWITCH_CLIENT_ID', 'TWITCH_CLIENT_SECRET', 'YOUTUBE_API_KEY'].includes(k)) {
            await env.OMNIMARK_KV.put(k, v);
          }
        }
        return new Response(JSON.stringify({ success: true }), { headers: corsHeaders });
      }

      const tId = isAuthorized ? (await getConfig(env, 'TWITCH_CLIENT_ID') || '') : '';
      const tSec = isAuthorized ? (await getConfig(env, 'TWITCH_CLIENT_SECRET') || '') : '';
      const yKey = isAuthorized ? (await getConfig(env, 'YOUTUBE_API_KEY') || '') : '';

      const html = `<html><head><meta charset="UTF-8"></head><body style="background:#121212;color:#eee;font-family:sans-serif;padding:20px">
        <div style="max-width:600px;margin:0 auto;background:#1e1e1e;padding:30px;border-radius:12px;border:1px solid #333">
          <h1 style="color:#2196f3;margin-top:0">OmniMark Admin</h1>
          <div style="margin-bottom:20px">
            <label>Clé API (pour accéder aux données et sauvegarder)</label>
            <div style="display:flex;gap:10px;margin-top:5px">
              <input type="password" id="key" value="${apiKey || ''}" style="flex:1;padding:10px;background:#121212;border:1px solid #444;color:#fff;border-radius:4px">
              <button onclick="location.href='/admin?key='+document.getElementById('key').value" style="padding:10px;background:#333;border:1px solid #444;color:#fff;border-radius:4px;cursor:pointer">Accéder</button>
            </div>
          </div>
          ${!isAuthorized ? '<p style="color:#f44336;font-weight:bold;text-align:center">⚠️ Clé API invalide ou manquante. Les données actuelles sont masquées.</p>' : ''}
          <div style="display:grid;gap:15px;${!isAuthorized ? 'opacity:0.5;pointer-events:none' : ''}">
            <div><label>Twitch Client ID</label><input id="TWITCH_CLIENT_ID" value="${tId}" style="width:100%;padding:10px;margin-top:5px;background:#121212;border:1px solid #444;color:#fff;border-radius:4px"></div>
            <div><label>Twitch Client Secret</label><input type="password" id="TWITCH_CLIENT_SECRET" value="${tSec}" style="width:100%;padding:10px;margin-top:5px;background:#121212;border:1px solid #444;color:#fff;border-radius:4px"></div>
            <div><label>YouTube API Key (Optionnel)</label><input type="password" id="YOUTUBE_API_KEY" value="${yKey}" placeholder="Laisser vide pour utiliser le flux RSS (méthode actuelle)" style="width:100%;padding:10px;margin-top:5px;background:#121212;border:1px solid #444;color:#fff;border-radius:4px"></div>
            <button onclick="save()" style="padding:12px;background:#2196f3;border:0;color:#fff;border-radius:4px;cursor:pointer;font-weight:bold">Sauvegarder la configuration</button>
            <button onclick="refresh()" id="btn-refresh" style="padding:12px;background:#4caf50;border:0;color:#fff;border-radius:4px;cursor:pointer;font-weight:bold">Rafraîchir le cache maintenant</button>
            <div id="msg" style="text-align:center;font-weight:bold;margin-top:10px"></div>
          </div>
          <hr style="border:0;border-top:1px solid #333;margin:20px 0">
          <p style="font-size:12px;color:#888">Note : La liste des chaînes est automatiquement synchronisée depuis l'extension.</p>
        </div>
        <script>
          const msg = document.getElementById('msg');
          async function save() {
            const key = document.getElementById('key').value;
            const data = {
              TWITCH_CLIENT_ID: document.getElementById('TWITCH_CLIENT_ID').value,
              TWITCH_CLIENT_SECRET: document.getElementById('TWITCH_CLIENT_SECRET').value,
              YOUTUBE_API_KEY: document.getElementById('YOUTUBE_API_KEY').value
            };
            const r = await fetch('/admin?key='+key, { method: 'POST', body: JSON.stringify(data) });
            msg.innerText = r.ok ? '✅ Sauvegardé !' : '❌ Erreur (Vérifiez la clé API)';
            msg.style.color = r.ok ? '#53fc18' : '#f44336';
            if (r.ok) setTimeout(() => location.reload(), 1000);
          }
          async function refresh() {
            const key = document.getElementById('key').value;
            const btn = document.getElementById('btn-refresh');
            btn.disabled = true; btn.innerText = 'Chargement...';
            const r = await fetch('/refresh?key='+key);
            msg.innerText = r.ok ? '✅ Cache actualisé !' : '❌ Erreur lors du rafraîchissement';
            msg.style.color = r.ok ? '#53fc18' : '#f44336';
            btn.disabled = false; btn.innerText = 'Rafraîchir le cache maintenant';
          }
        </script>
      </body></html>`;
      return new Response(html, { headers: { ...corsHeaders, 'Content-Type': 'text/html;charset=UTF-8' } });
    }

    if (path === '/config/channels') {
      if (request.method !== 'POST') return new Response("Method not allowed", { status: 405 });
      if (env.API_KEY && apiKey !== env.API_KEY) return new Response("Unauthorized", { status: 401 });
      const data = await request.json();
      if (data.twitch) await env.OMNIMARK_KV.put('CHANNELS_TWITCH', JSON.stringify(data.twitch));
      if (data.youtube) await env.OMNIMARK_KV.put('CHANNELS_YOUTUBE', JSON.stringify(data.youtube));
      ctx.waitUntil(refreshCache(env));
      return new Response(JSON.stringify({ success: true }), { headers: corsHeaders });
    }

    if (env.API_KEY && apiKey !== env.API_KEY) return new Response(JSON.stringify({ error: "Unauthorized" }), { status: 401, headers: corsHeaders });
    
    if (path === '/test') return new Response(JSON.stringify({ status: "ok" }), { headers: corsHeaders });
    
    if (path === '/refresh') {
      ctx.waitUntil(refreshCache(env));
      return new Response(JSON.stringify({ success: true }), { headers: corsHeaders });
    }

    // --- YouTube API ---
    if (path === "/youtube") {
      const { handle, id } = Object.fromEntries(url.searchParams);
      const ua = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/110.0.0.0 Safari/537.36";
      
      if (handle) {
        const target = handle.startsWith("@") ? handle : "@" + handle;
        const res = await fetch(`https://www.youtube.com/${target}`, { headers: { "User-Agent": ua } });
        const html = await res.text();
        const cid = html.match(/channelId":"(UC[^"]+)"/)?.[1] || 
                    html.match(/externalId":"(UC[^"]+)"/)?.[1] ||
                    html.match(/browseId":"(UC[^"]+)"/)?.[1] ||
                    html.match(/itemprop="identifier" content="(UC[^"]+)"/)?.[1] ||
                    html.match(/rel="canonical" href="https:\/\/www\.youtube\.com\/channel\/(UC[^"]+)"/)?.[1];
        const name = html.match(/<meta property="og:title" content="([^"]+)">/)?.[1] || 
                     html.match(/"name":"([^"]+)"/)?.[1] || handle;
        return new Response(JSON.stringify({ channelId: cid, name }), { headers: corsHeaders });
      }
      if (id) {
        const res = await fetch(`https://www.youtube.com/channel/${id}`, { headers: { "User-Agent": ua } });
        const html = await res.text();
        const nameMatch = html.match(/<meta property="og:title" content="([^"]+)">/)?.[1] || 
                          html.match(/<title>([^<]+) - YouTube<\/title>/)?.[1];
        return new Response(JSON.stringify({ name: nameMatch || id }), { headers: corsHeaders });
      }
      
      const cached = await env.OMNIMARK_KV?.get("CACHE_YOUTUBE");
      return new Response(cached || "[]", { headers: corsHeaders });
    }

    // --- Twitch/Kick API ---
    if (path === '/streams') {
      const cached = await env.OMNIMARK_KV?.get('CACHE_STREAMS');
      return new Response(cached || '[]', { headers: corsHeaders });
    }

    return new Response("Not Found", { status: 404, headers: corsHeaders });
  },

  async scheduled(event, env, ctx) {
    await refreshCache(env);
  }
};
